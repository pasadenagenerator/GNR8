import { useCallback, useEffect, useState } from 'react';
import GitChip from '../../../vendor/stacki/src/features/git/GitChip';
import { readGitInfo } from '../../../vendor/stacki/src/features/git/gitChipBridge';
import { useGitIncoming } from '../../../vendor/stacki/src/features/git/useGitIncoming';
import HistoryTimeline from '../../../vendor/stacki/src/features/history/HistoryTimeline';
import { HistorySection } from '../../../vendor/stacki/src/features/history/HistorySections';
import { createPanelCache, usePanelData } from '../../../vendor/stacki/src/lib/usePanelData';
import { PanelReadStatus } from '../../../vendor/stacki/src/ui/PanelReadStatus';
import type { IpcResults } from '../../../vendor/stacki/shared/ipc/ipcResults';
import type { Project } from '../project/projectClient';
import { requestGnr8GitHistory } from '../project/Gnr8Save';

const cache = createPanelCache<IpcResults['git:info']>();
const noPreview = () => {};
function subscribe(refresh: () => void) {
  window.addEventListener('studio:project-changed', refresh);
  window.addEventListener('studio:source-updated', refresh);
  return () => {
    window.removeEventListener('studio:project-changed', refresh);
    window.removeEventListener('studio:source-updated', refresh);
  };
}
type Props = { readonly project: Project; readonly flushSave: () => Promise<unknown>;
  readonly refreshProject: () => Promise<unknown>; readonly report: (error: unknown) => void };
export function GitPanel(props: Props) {
  const [mode,setMode]=useState<'loading'|'github'|'native'>('loading');
  useEffect(() => {
    let active=true;
    void fetch('/studio/api/gnr8-persistence').then(response => response.json()).then(config => {
      if(active)setMode(config.storage==='github'?'github':'native');
    }).catch(() => {if(active)setMode('native');});
    return () => {active=false;};
  },[]);
  return mode==='github' ? <Gnr8GitPanel /> : mode==='native' ? <NativeGitPanel {...props} /> :
    <div className="panel-section grow"><div className="panel-header"><h2>Git</h2></div><p>Loading history…</p></div>;
}
function Gnr8GitPanel() {
  const [history,setHistory]=useState<Array<{revision:number;sha256:string;commit:string}>>([]);
  const [error,setError]=useState('');
  const [loading,setLoading]=useState(true);
  const refresh=useCallback(() => {
    setLoading(true);setError('');
    void requestGnr8GitHistory().then(entries => setHistory(entries)).catch(reason =>
      setError(reason instanceof Error?reason.message:'GitHub history is unavailable.')).finally(() => setLoading(false));
  },[]);
  useEffect(() => {
    refresh();window.addEventListener('gnr8:save-confirmed',refresh);
    return () => window.removeEventListener('gnr8:save-confirmed',refresh);
  },[refresh]);
  return <div className="panel-section grow">
    <div className="panel-header"><h2>Git</h2></div>
    <div className="panel-body history-body">
      <p>Save creates a revision in this site's private GitHub branch. Undo and Redo work in the open editor.</p>
      <button type="button" onClick={refresh} disabled={loading}>Retry</button>
      {loading && <p role="status">Loading GitHub history…</p>}
      {error && <p role="alert">{error}</p>}
      {!loading && !error && <ol>{history.map(entry => <li key={entry.commit}>
        <strong>Revision {entry.revision}</strong> · commit <code>{entry.commit.slice(0,7)}</code>
      </li>)}</ol>}
    </div>
  </div>;
}
function NativeGitPanel({
  project,
  flushSave,
  refreshProject,
  report,
}: {
  readonly project: Project;
  readonly flushSave: () => Promise<unknown>;
  readonly refreshProject: () => Promise<unknown>;
  readonly report: (error: unknown) => void;
}) {
  const key = project.workspaceId ?? project.name;
  const [open, setOpen] = useState(true);
  const [notice, setNotice] = useState('');
  const read = useCallback(() => readGitInfo('studio'), []);
  const { state, refresh } = usePanelData(cache, key, read, subscribe);
  const info = state.kind === 'ready' ? state.value : undefined;
  const incoming = useGitIncoming('studio', info);
  return (
    <div className="panel-section grow">
      <div className="panel-header">
        <h2>Git</h2>
      </div>
      <div className="panel-body history-body">
        <PanelReadStatus state={state} label="Git history" layout="history" retry={refresh} />
        {info && (
          <GitChip
            info={info}
            refresh={refresh}
            busy={undefined}
            project={{ path: 'studio', name: project.name }}
            incoming={incoming}
            flushSave={flushSave}
            onWorktreeChanged={refreshProject}
            showToast={(message, kind) => (kind === 'error' ? report(message) : setNotice(message))}
          />
        )}
        {notice && (
          <p role="status" className="history-note">
            {notice}
          </p>
        )}
        {info?.isRepo && (
          <HistorySection title="Source Control Graph" open={open} onToggle={() => setOpen(!open)}>
            <HistoryTimeline
              projectPath={key}
              branch={info.branch}
              branches={info.branches}
              trunk={info.trunk ?? info.branch}
              head={info.head}
              userEmail={info.userEmail}
              previewRef={undefined}
              onExitPreview={noPreview}
            />
          </HistorySection>
        )}
      </div>
    </div>
  );
}
